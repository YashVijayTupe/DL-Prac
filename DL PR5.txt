# Practical 5: Continuous Bag of Words (CBOW) Model

# a. Data preparation
import numpy as np
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Embedding, Dense, GlobalAveragePooling1D
from tensorflow.keras.preprocessing.text import Tokenizer

text = """
deep learning is a part of machine learning machine learning is a part of artificial intelligence deep learning uses neural networks
"""

tokenizer = Tokenizer()
tokenizer.fit_on_texts([text])

word_index = tokenizer.word_index
vocab_size = len(word_index) + 1

# Convert text into sequences
sequence = tokenizer.texts_to_sequences([text])[0]

# b. Generate training data
window_size = 2
X = []
y = []

for i in range(window_size, len(sequence) - window_size):
    context = (
        sequence[i-window_size:i] +
        sequence[i+1:i+window_size+1]
    )
    target = sequence[i]

    X.append(context)
    y.append(target)

X = np.array(X)
y = np.array(y)

# c. Train CBOW model
model = Sequential([
    Embedding(vocab_size, 50, input_length=window_size * 2),
    GlobalAveragePooling1D(),
    Dense(vocab_size, activation='softmax')
])

model.compile(
    optimizer='adam',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

model.fit(X, y, epochs=100, verbose=0)

# d. Output: Predict target word
context_words = ["deep", "is", "part", "machine"]

context_ids = [word_index[word] for word in context_words]
context_input = np.array([context_ids])

prediction = model.predict(context_input, verbose=0)
predicted_id = np.argmax(prediction)

predicted_word = ""

for word, index in word_index.items():
    if index == predicted_id:
        predicted_word = word
        break

print("Context Words:", context_words)
print("Predicted Target Word:", predicted_word)
print("CBOW Model trained successfully!")
